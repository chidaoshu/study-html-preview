3.5.1行を並べ替える必修

構文
SELECT 列リスト FROM 表名 [ WHERE 条件 ] ORDER BY 列名|列別名|列番号 [ ASC|DESC ] [ NULLS FIRST|NULLS LAST ] ;
キーの指定 — 3 つとも同じ結果
ORDER BY に何を書くか 
SELECT ename, sal FROM emp
ORDER BY ⟦sal⟧ ;
基本形
SELECT ename, sal ⟦AS salary⟧ FROM emp
ORDER BY ⟦salary⟧ ;
SELECT 句で付けた別名が使える
SELECT ename, sal FROM emp
ORDER BY ⟦2⟧ ;
SELECT 句リストの何番目か（sal ＝ 2）

ORDER BY 列名 [ASC|DESC] [ OFFSET m ROWS ] FETCH FIRST n [PERCENT] ROWS ONLY|WITH TIES ;

emp から 5 人分

ORDER BY sal FETCH FIRST 5 ROWS ONLY;
